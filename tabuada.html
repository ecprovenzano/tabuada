<script>

const TOTAL_Q=10;

let selected=new Set(),score,lives,current,blocked,questions,correctAnswer;

let elapsed=0,timerInterval=null;

let scores=[];


function saveScores(){try{localStorage.setItem('tabuada-mario-scores',JSON.stringify(scores));}catch(e){}}

function loadScores(){try{const r=localStorage.getItem('tabuada-mario-scores');if(r)scores=JSON.parse(r);}catch(e){}renderLB();updateHeader();}

function fmtTime(s){return Math.floor(s/60)+':'+(s%60<10?'0':'')+s%60;}


function renderLB(){

  const html=scores.length?scores.map((s,i)=>`

    <div class="lb-row${i===0?' lb-best':''}">

      <span class="lb-rank">${i+1}.</span>

      <span class="lb-tables">${s.tables}</span>

      <span class="lb-score">${s.score}pts</span>

      <span class="lb-time">${fmtTime(s.time)}</span>

    </div>`).join(''):'<div class="lb-empty">nenhum recorde ainda</div>';

  document.getElementById('lb-rows').innerHTML=html;

  const el2=document.getElementById('lb-rows2');

  if(el2)el2.innerHTML=html;

}


function tabKey(){return[...selected].sort((a,b)=>a-b).join(',');}


function updateHeader(){

  const tabs=tabKey();

  document.getElementById('hdr-tables').textContent=tabs||'—';

  const rel=scores.filter(s=>s.tables===tabs);

  document.getElementById('hdr-best').textContent=rel.length?rel[0].score+'pts':'—';

}


function updateSelSummary(){

  const arr=[...selected].sort((a,b)=>a-b);

  const el=document.getElementById('sel-summary');

  el.textContent=arr.length?'selecionado: tabuada do '+arr.join(', '):'nenhuma selecionada';

  updateHeader();

}


const grid=document.getElementById('tables-grid');

for(let n=1;n<=10;n++){

  const btn=document.createElement('button');

  btn.className='tbl-btn';btn.textContent=n;btn.id='tb'+n;

  btn.onclick=()=>{

    if(selected.has(n)){selected.delete(n);btn.classList.remove('sel');}

    else{selected.add(n);btn.classList.add('sel');}

    document.getElementById('start-btn').disabled=selected.size===0;

    updateSelSummary();

  };

  grid.appendChild(btn);

}


function buildProgress(){

  const pw=document.getElementById('prog-wrap');pw.innerHTML='';

  for(let i=0;i<TOTAL_Q;i++){const d=document.createElement('div');d.className='prog-coin';d.id='pc'+i;pw.appendChild(d);}

}


function shuffle(arr){for(let i=arr.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[arr[i],arr[j]]=[arr[j],arr[i]];}return arr;}


function generateQuestions(){

  let pool=[];

  selected.forEach(t=>{for(let i=1;i<=10;i++)pool.push({a:t,b:i,ans:t*i});});

  return shuffle(pool).slice(0,TOTAL_Q);

}


function showMenu(){

  clearInterval(timerInterval);

  ['end-screen','play-screen'].forEach(id=>document.getElementById(id).classList.remove('active'));

  document.getElementById('menu-screen').classList.add('active');

  renderLB();

}


function startGame(){

  score=0;lives=3;current=0;blocked=false;elapsed=0;

  questions=generateQuestions();

  clearInterval(timerInterval);

  timerInterval=setInterval(()=>{elapsed++;document.getElementById('hv-time').textContent=fmtTime(elapsed);},1000);

  buildProgress();

  ['menu-screen','end-screen'].forEach(id=>document.getElementById(id).classList.remove('active'));

  document.getElementById('play-screen').classList.add('active');

  updateHUD();loadQuestion();

}


function livesStr(n){return Array(3).fill(0).map((_,i)=>i<n?'♥':'♡').join('');}


function updateHUD(){

  document.getElementById('hv-score').textContent=score;

  document.getElementById('hv-lives').textContent=livesStr(lives);

  document.getElementById('hv-q').textContent=(current+1)+'/'+TOTAL_Q;

}


function loadQuestion(){

  if(current>=TOTAL_Q){clearInterval(timerInterval);endGame(true);return;}

  blocked=false;

  const q=questions[current];

  correctAnswer=q.ans;

  document.getElementById('question-text').textContent=q.a+' × '+q.b+' = ?';

  document.getElementById('feedback').textContent='';

  const inp=document.getElementById('answer-input');

  inp.value='';inp.className='';

  document.getElementById('hv-q').textContent=(current+1)+'/'+TOTAL_Q;

  setTimeout(()=>inp.focus(),80);

}


document.getElementById('answer-input').addEventListener('keydown',e=>{if(e.key==='Enter')confirmAnswer();});


function confirmAnswer(){

  if(blocked)return;

  const inp=document.getElementById('answer-input');

  const val=parseInt(inp.value);

  if(isNaN(val)){document.getElementById('feedback').textContent='DIGITE UM NÚMERO!';return;}

  blocked=true;

  if(val===correctAnswer){

    inp.classList.add('correct');score+=10;

    document.getElementById('feedback').textContent='CORRETO! +10';

    document.getElementById('pc'+current).classList.add('done');

    updateHUD();

    setTimeout(()=>{current++;loadQuestion();},700);

  }else{

    inp.classList.add('wrong');lives--;

    document.getElementById('pc'+current).classList.add('wrong-mark');

    document.getElementById('feedback').textContent='ERA '+correctAnswer+'!';

    updateHUD();

    if(lives<=0){setTimeout(()=>{clearInterval(timerInterval);endGame(false);},1000);}

    else{setTimeout(()=>{current++;loadQuestion();},1200);}

  }

}


function endGame(win){

  document.getElementById('play-screen').classList.remove('active');

  const tabs=tabKey();

  const entry={score,time:elapsed,tables:tabs,date:new Date().toLocaleDateString('pt-BR')};

  const prevBest=scores.filter(s=>s.tables===tabs)[0];

  const isRecord=!prevBest||(score>prevBest.score)||(score===prevBest.score&&elapsed<prevBest.time);

  scores.push(entry);

  scores.sort((a,b)=>b.score-a.score||(a.time-b.time));

  scores=scores.slice(0,10);

  saveScores();


  document.getElementById('end-title').textContent=win?'YOU WIN!':'GAME OVER';

  const acertos=score/10;

  document.getElementById('end-stats').innerHTML=`

    <div class="es-row"><span class="es-lbl">PONTUAÇÃO</span><span class="es-val">${score} pts</span></div>

    <div class="es-row"><span class="es-lbl">ACERTOS</span><span class="es-val">${acertos}/${TOTAL_Q}</span></div>

    <div class="es-row"><span class="es-lbl">ERROS</span><span class="es-val">${TOTAL_Q-acertos}</span></div>

    <div class="es-row"><span class="es-lbl">TEMPO</span><span class="es-val">${fmtTime(elapsed)}</span></div>

    <div class="es-row"><span class="es-lbl">TABUADAS</span><span class="es-val">${tabs}</span></div>

  `;

  const recEl=document.getElementById('new-rec-msg');

  if(isRecord&&win){

    recEl.innerHTML='<div class="new-record">★ NOVO RECORDE! ★<br>'+score+' pts em '+fmtTime(elapsed)+'</div>';

  }else if(prevBest&&win){

    const diff=score-prevBest.score;

    recEl.innerHTML='<div style="color:#aaa;font-size:8px;text-align:center;margin-bottom:10px;line-height:2;">recorde: '+prevBest.score+' pts / '+fmtTime(prevBest.time)+'<br>'+(diff>=0?'+'+diff+' pts vs recorde':'ainda não chegou lá!')+'</div>';

  }else{recEl.innerHTML='';}


  renderLB();updateHeader();

  document.getElementById('end-screen').classList.add('active');

}


loadScores();

</script>
